import json, re, subprocess

def run(args, display=True):
    result = subprocess.run(args, text=True, capture_output=True, timeout=40)
    if display:
        print('\nCOMMAND ' + json.dumps(args))
        print(result.stdout.strip())
        print('EXIT ' + str(result.returncode))
    return result

def emit(label, value):
    print(label + ' ' + json.dumps(value, sort_keys=True))

run(['date', '-Is'])
run(['sudo', '-n', '/bin/true'])
run(['sudo', '-n', 'docker', 'version'])
